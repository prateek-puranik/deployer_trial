# Use a slim Python image for a smaller final image size
FROM python:3.9-slim-buster

# Set the working directory inside the container
WORKDIR /app

# Copy the requirements file into the container
COPY requirements.txt .

# Install Python dependencies, including uvicorn
# --no-cache-dir: Prevents pip from saving downloaded packages, reducing image size
# --upgrade pip: Ensures pip is up-to-date
RUN pip install --no-cache-dir --upgrade pip && \
    pip install --no-cache-dir -r requirements.txt

# Copy the rest of your application code into the container
# This includes main.py, auth_utils.py, models.py, database.py, etc.
COPY . .

# Cloud Run expects applications to listen on the port specified by the PORT environment variable.
# Default to 8080 if not set.
ENV PORT 8080

# Expose the port the application will run on
EXPOSE $PORT

# Command to run the application using Uvicorn
# 'main:app' refers to the 'app' object in 'main.py'
# --host 0.0.0.0: Binds to all network interfaces, making it accessible from outside the container
# --port $PORT: Uses the PORT environment variable
# --workers 1: For Cloud Run, a single worker is usually sufficient as Cloud Run handles scaling instances.
CMD ["uvicorn", "image_text_processor:app", "--host", "0.0.0.0", "--port", "$PORT", "--workers", "1"]
